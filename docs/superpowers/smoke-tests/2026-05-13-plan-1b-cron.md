## Manual job invocation

Jobs can be triggered immediately without waiting for cron ticks:

```bash
# Run from apps/api/ directory

# daily-refresh
pnpm exec tsx -e "
import { loadEnv } from './src/env.js';
import { runDailyRefresh } from './src/cron/jobs/daily-refresh.js';
const env = loadEnv();
await runDailyRefresh(env).then(() => process.exit(0));
"

# retry-unprocessed
pnpm exec tsx -e "
import { loadEnv } from './src/env.js';
import { runRetryUnprocessed } from './src/cron/jobs/retry-unprocessed.js';
const env = loadEnv();
await runRetryUnprocessed(env).then(() => process.exit(0));
"

# cleanup-events (use TTL=1 so any day-old row is cleaned)
CRON_EVENT_TTL_DAYS=1 pnpm exec tsx -e "
import { loadEnv } from './src/env.js';
import { runCleanupEvents } from './src/cron/jobs/cleanup-events.js';
const env = loadEnv();
await runCleanupEvents(env).then(() => process.exit(0));
"
```

## Part 1: daily-refresh

```bash
# Check existing active items
docker exec fina-postgres-dev psql -U fina -d fina -c \
  "SELECT id, \"institutionName\", status FROM plaid_items WHERE status = 'active';"

# Run daily-refresh (from apps/api/)
# Expected API log: "daily-refresh complete" { itemsTotal: N, synced: M, gated: 0, lockHeld: 0, failed: 0 }

# Verify lastSyncAt updated for synced items
docker exec fina-postgres-dev psql -U fina -d fina -c \
  "SELECT id, \"lastSyncAt\" FROM plaid_items WHERE status = 'active' ORDER BY \"lastSyncAt\" DESC LIMIT 3;"
```

## Part 2: retry-unprocessed

```bash
# Seed a NULL processedAt event (simulate a lock_held event)
docker exec fina-postgres-dev psql -U fina -d fina -c "
INSERT INTO plaid_webhook_events (id, \"plaidItemId\", \"webhookType\", \"webhookCode\", body, \"receivedAt\")
SELECT gen_random_uuid(), id, 'TRANSACTIONS', 'SYNC_UPDATES_AVAILABLE', '{}',
  NOW() - INTERVAL '5 minutes'
FROM plaid_items WHERE status = 'active' LIMIT 1;
"

# Run retry-unprocessed
# Expected log: "retry-unprocessed complete" { found: 1, attempted: 1, gated: 0 }

# Verify row was processed
docker exec fina-postgres-dev psql -U fina -d fina -c \
  "SELECT id, \"webhookCode\", \"processedAt\", \"processError\" FROM plaid_webhook_events ORDER BY \"receivedAt\" DESC LIMIT 3;"
```

## Part 3: cleanup-events (TTL)

```bash
# Seed an old event manually
docker exec fina-postgres-dev psql -U fina -d fina -c "
INSERT INTO plaid_webhook_events (id, \"webhookType\", \"webhookCode\", body, \"receivedAt\")
VALUES (gen_random_uuid(), 'TRANSACTIONS', 'SYNC_UPDATES_AVAILABLE', '{}',
  NOW() - INTERVAL '31 days');
"

BEFORE=$(docker exec fina-postgres-dev psql -U fina -d fina -tAc \
  "SELECT count(*) FROM plaid_webhook_events;")

# Run cleanup with 30-day TTL
CRON_EVENT_TTL_DAYS=30 pnpm exec tsx -e "
import { loadEnv } from './src/env.js';
import { runCleanupEvents } from './src/cron/jobs/cleanup-events.js';
const env = loadEnv();
await runCleanupEvents(env).then(() => process.exit(0));
"

AFTER=$(docker exec fina-postgres-dev psql -U fina -d fina -tAc \
  "SELECT count(*) FROM plaid_webhook_events;")
echo "before=$BEFORE after=$AFTER (should differ by 1)"
```

## Part 4: tier gate

```bash
# Expire a user's trial via DB
docker exec fina-postgres-dev psql -U fina -d fina -c "
UPDATE users SET \"trialEndsAt\" = NOW() - INTERVAL '1 day', \"trialConsumed\" = true
WHERE email = 'your-test-email@example.com';
"

# Run daily-refresh
# Expected log: item for that user → gated=1

# Restore trial
docker exec fina-postgres-dev psql -U fina -d fina -c "
UPDATE users SET \"trialEndsAt\" = NOW() + INTERVAL '7 days', \"trialConsumed\" = false
WHERE email = 'your-test-email@example.com';
"
```

## Exit Criteria

- [ ] API log shows 3 scheduler registrations on start with `CRON_ENABLED=true`
- [ ] `daily-refresh` manual invocation logs `synced = N` for active items
- [ ] `retry-unprocessed` stamps `processedAt` on a seeded NULL event
- [ ] `cleanup-events` deletes rows older than TTL, keeps recent ones
- [ ] Tier-gated user's items are logged as `gated = 1`, no sync attempted
