# Plan 1E-chat-foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** End-to-end Pro-only chat agent that calls 4 representative tools (3 read + 1 safe write) against Gemini 2.5 Flash Lite, persisting one workspace-scoped adhoc thread.

**Architecture:** Adds 2 tables (`chat_threads`, `chat_messages`) and a thin per-tool registry; tool dispatcher returns `{ ok, result } | { ok, error }` (never throws) and a `processChatTurn` engine drives the LLM ↔ tool loop bounded by `MAX_TOOL_CALLS`. The existing inline manual-categorize algorithm in `apps/api/src/routes/transactions.ts:126-246` is **extracted first** into `apps/api/src/categorize/manual-categorize.ts` so both the HTTP route AND the new chat tool share one source of truth.

**Tech Stack:** TypeScript, Fastify, Prisma + PostgreSQL, Zod, Vitest, Playwright (E2E), `@google/genai` (already integrated), `zod-to-json-schema` (new dep).

**Spec reference:** `docs/superpowers/specs/2026-05-14-plan-1e-chat-foundation-design.md`

```ts
import { prisma } from "../db.js";
import { normalizeMerchant } from "./normalize.js";

export class ManualCategorizeError extends Error {
  constructor(public code: "tx_not_found" | "category_not_found") {
    super(code);
    this.name = "ManualCategorizeError";
  }
}

export type ManualCategorizeInput = {
  workspaceId: string;
  txId: string;
  categoryName: string;
};

export type ManualCategorizeResult = {
  tx: {
    id: string;
    amount: string;
    currency: string;
    merchantRaw: string;
    merchantNormalized: string | null;
    categoryId: string | null;
    categorizationSource: "manual" | "mapping" | "ai" | null;
    occurredAt: Date;
  };
  backfilledCount: number;
};

export async function applyManualCategorize(
  input: ManualCategorizeInput,
): Promise<ManualCategorizeResult> {
  const trimmedName = input.categoryName.trim();

  const tx = await prisma.transaction.findFirst({
    where: { id: input.txId, workspaceId: input.workspaceId },
    select: { id: true, merchantRaw: true, merchantNormalized: true },
  });
  if (!tx) throw new ManualCategorizeError("tx_not_found");

  const merchantNorm = tx.merchantNormalized ?? normalizeMerchant(tx.merchantRaw);

  const category = await prisma.category.findFirst({
    where: {
      workspaceId: input.workspaceId,
      archivedAt: null,
      name: { equals: trimmedName, mode: "insensitive" },
    },
    select: { id: true, name: true },
  });
  if (!category) throw new ManualCategorizeError("category_not_found");

  const result = await prisma.$transaction(async (db) => {
    await db.transaction.update({
      where: { id: tx.id },
      data: {
        merchantNormalized: merchantNorm,
        categoryId: category.id,
        categorizationSource: "manual",
      },
    });

    await db.merchantMapping.upsert({
      where: {
        workspaceId_merchantNormalized: {
          workspaceId: input.workspaceId,
          merchantNormalized: merchantNorm,
        },
      },
      create: {
        workspaceId: input.workspaceId,
        merchantNormalized: merchantNorm,
        categoryId: category.id,
        confidence: 1.0,
        source: "manual",
      },
      update: { categoryId: category.id, source: "manual", confidence: 1.0 },
    });

    // Normalize null-merchantNormalized siblings.
    const nullSiblings = await db.transaction.findMany({
      where: {
        workspaceId: input.workspaceId,
        merchantNormalized: null,
        merchantRaw: {
          mode: "insensitive",
          contains: trimmedName.length >= 3 ? merchantNorm.slice(0, 3) : merchantNorm,
        },
      },
      select: { id: true, merchantRaw: true },
    });
    for (const ns of nullSiblings) {
      const norm = normalizeMerchant(ns.merchantRaw);
      if (norm === merchantNorm) {
        await db.transaction.update({ where: { id: ns.id }, data: { merchantNormalized: norm } });
      }
    }

    const backfilled = await db.transaction.updateMany({
      where: {
        workspaceId: input.workspaceId,
        merchantNormalized: merchantNorm,
        id: { not: tx.id },
        OR: [
          { categorizationSource: null },
          { categorizationSource: { not: "manual" } },
        ],
      },
      data: { categoryId: category.id, categorizationSource: "mapping" },
    });

    return { backfilledCount: backfilled.count };
  });

  const updated = await prisma.transaction.findUniqueOrThrow({
    where: { id: tx.id },
    select: {
      id: true, amount: true, currency: true, merchantRaw: true, merchantNormalized: true,
      categoryId: true, categorizationSource: true, occurredAt: true,
    },
  });

  return {
    tx: {
      id: updated.id,
      amount: updated.amount.toString(),
      currency: updated.currency,
      merchantRaw: updated.merchantRaw,
      merchantNormalized: updated.merchantNormalized,
      categoryId: updated.categoryId,
      categorizationSource: updated.categorizationSource,
      occurredAt: updated.occurredAt,
    },
    backfilledCount: result.backfilledCount,
  };
}
```

    const params = req.params as { id: string; txId: string };
    const parsed = CategorizeBody.safeParse(req.body);
    if (!parsed.success) return reply.code(400).send({ error: "invalid_body" });
    try {
      const ctx = await requireWorkspaceContext(req, app, { requiredRole: "editor" });
      if (ctx.membership.workspaceId !== params.id) {

      }
      const result = await applyManualCategorize({
        workspaceId: ctx.membership.workspaceId,
        txId: params.txId,
        categoryName: parsed.data.categoryName,
      });
      return {
        transaction: {
          ...result.tx,
          amount: result.tx.amount,
          occurredAt: result.tx.occurredAt.toISOString(),
        },
        backfilledCount: result.backfilledCount,
      };
    } catch (e) {
      if (e instanceof ManualCategorizeError) {
        if (e.code === "tx_not_found") return reply.code(404).send({ error: "not_found" });
        if (e.code === "category_not_found") return reply.code(404).send({ error: "category_not_found" });
      }
      if (e instanceof HttpError) return reply.code(e.statusCode).send({ error: e.code });
      throw e;
    }
  });

import type { ToolContext, ToolDef } from "./tools/types.js";

export type DispatchResult =
  | { ok: true; result: unknown }
  | { ok: false; error: string };

export async function dispatchTool(
  ctx: ToolContext,
  toolName: string,
  rawArgs: unknown,
  registry: ToolDef[],
): Promise<DispatchResult> {
  const tool = registry.find((t) => t.name === toolName);
  if (!tool) return { ok: false, error: `unknown_tool: ${toolName}` };

  // Gemini may omit `args` for zero-arg tools — coerce so z.object({}) parses.
  const parsed = tool.args.safeParse(rawArgs ?? {});
  if (!parsed.success) return { ok: false, error: `invalid_args: ${parsed.error.message}` };

  try {
    const result = await tool.run(ctx, parsed.data);
    return { ok: true, result };
  } catch (e) {
    const msg = e instanceof Error ? e.message : String(e);
    console.error("chat tool failed", { toolName, err: e });
    return { ok: false, error: msg.slice(0, 500) };
  }
}

import { z } from "zod";
import { prisma } from "../../db.js";
import type { ToolDef } from "./types.js";

const Args = z.object({
  period: z
    .enum(["today", "this_week", "this_month", "last_month"])
    .describe("Time window for the listing. Omit for all-time.")
    .optional(),
  limit: z
    .number()
    .int()
    .positive()
    .max(50)
    .describe("Max transactions to return. Default 20, hard cap 50.")
    .optional(),
});

type Args = z.infer<typeof Args>;

function periodRange(period: NonNullable<Args["period"]>): { gte: Date; lt: Date } {
  const now = new Date();
  const y = now.getFullYear();
  const m = now.getMonth();
  const d = now.getDate();
  if (period === "today") {
    const start = new Date(y, m, d);
    const end = new Date(y, m, d + 1);
    return { gte: start, lt: end };
  }
  if (period === "this_week") {
    const day = now.getDay(); // 0=Sun
    const monday = new Date(y, m, d - ((day + 6) % 7));
    return { gte: monday, lt: new Date(y, m, monday.getDate() + 7) };
  }
  if (period === "this_month") {
    return { gte: new Date(y, m, 1), lt: new Date(y, m + 1, 1) };
  }
  // last_month
  return { gte: new Date(y, m - 1, 1), lt: new Date(y, m, 1) };
}

export const listTransactions: ToolDef<Args, {
  transactions: Array<{
    id: string;
    amount: string;
    merchantRaw: string;
    merchantNormalized: string | null;
    categoryId: string | null;
    occurredAt: string;
  }>;
}> = {
  name: "listTransactions",
  description: "List the user's recent transactions, optionally filtered by time period.",
  args: Args,
  async run(ctx, args) {
    const where: { workspaceId: string; removedAt: null; occurredAt?: { gte: Date; lt: Date } } = {
      workspaceId: ctx.workspaceId,
      removedAt: null,
    };
    if (args.period) where.occurredAt = periodRange(args.period);

    const rows = await prisma.transaction.findMany({
      where,
      orderBy: [{ occurredAt: "desc" }, { id: "desc" }],
      take: args.limit ?? 20,
      select: {
        id: true, amount: true, merchantRaw: true, merchantNormalized: true,
        categoryId: true, occurredAt: true,
      },
    });
    return {
      transactions: rows.map((r) => ({
        id: r.id,
        amount: r.amount.toString(),
        merchantRaw: r.merchantRaw,
        merchantNormalized: r.merchantNormalized,
        categoryId: r.categoryId,
        occurredAt: r.occurredAt.toISOString(),
      })),
    };
  },
};

import { z } from "zod";
import { Prisma } from "@prisma/client";
import { prisma } from "../../db.js";
import type { ToolDef } from "./types.js";

const Args = z.object({
  period: z.enum(["this_week", "this_month", "last_month"]).describe("Period to summarize."),
  group_by: z
    .enum(["category", "merchant", "day"])
    .describe("Bucket dimension. Defaults to 'category'.")
    .optional(),
});

type Args = z.infer<typeof Args>;

function periodRange(period: Args["period"]): { gte: Date; lt: Date } {
  const now = new Date();
  const y = now.getFullYear();
  const m = now.getMonth();
  if (period === "this_week") {
    const day = now.getDay();
    const monday = new Date(y, m, now.getDate() - ((day + 6) % 7));
    return { gte: monday, lt: new Date(y, m, monday.getDate() + 7) };
  }
  if (period === "this_month") return { gte: new Date(y, m, 1), lt: new Date(y, m + 1, 1) };
  return { gte: new Date(y, m - 1, 1), lt: new Date(y, m, 1) };
}

export const getSpendingSummary: ToolDef<Args, {
  period: Args["period"];
  groupBy: NonNullable<Args["group_by"]>;
  groups: Array<{ key: string; total: string; count: number }>;
}> = {
  name: "getSpendingSummary",
  description: "Total spending in a period bucketed by category, merchant, or day.",
  args: Args,
  async run(ctx, args) {
    const groupBy = args.group_by ?? "category";
    const { gte, lt } = periodRange(args.period);

    const rows = await prisma.transaction.findMany({
      where: {
        workspaceId: ctx.workspaceId,
        removedAt: null,
        occurredAt: { gte, lt },
      },
      select: { amount: true, categoryId: true, merchantNormalized: true, occurredAt: true },
    });

    const buckets = new Map<string, { total: Prisma.Decimal; count: number }>();
    for (const r of rows) {
      let key: string;
      if (groupBy === "category") key = r.categoryId ?? "uncategorized";
      else if (groupBy === "merchant") key = r.merchantNormalized ?? "unknown";
      else key = r.occurredAt.toISOString().slice(0, 10);

      const cur = buckets.get(key);
      if (cur) {
        cur.total = cur.total.plus(r.amount);
        cur.count += 1;
      } else {
        buckets.set(key, { total: new Prisma.Decimal(r.amount), count: 1 });
      }
    }
    const groups = [...buckets.entries()]
      .map(([key, v]) => ({ key, total: v.total.toFixed(2), count: v.count }))
      .sort((a, b) => parseFloat(b.total) - parseFloat(a.total));

    return { period: args.period, groupBy, groups };
  },
};

import { z } from "zod";
import { zodToJsonSchema } from "zod-to-json-schema";
import type { FunctionDeclaration } from "@google/genai";
import type { ToolDef } from "./tools/types.js";

export function toGeminiFunctionDeclarations(tools: ToolDef[]): FunctionDeclaration[] {
  return tools.map((t) => {
    const decl: FunctionDeclaration = { name: t.name, description: t.description };
    if (hasAnyKeys(t.args)) {
      decl.parametersJsonSchema = zodToJsonSchema(t.args, { $refStrategy: "none" });
    }
    return decl;
  });
}

function hasAnyKeys(schema: z.ZodType): boolean {
  if (!(schema instanceof z.ZodObject)) return true;
  return Object.keys(schema.shape).length > 0;
}

import { dispatchTool } from "./dispatcher.js";

import type { ToolContext } from "./tools/types.js";

const SYSTEM_PROMPT = `You are Fina, a personal-finance assistant for Canadian users. You help users
understand their spending, categorize transactions, and plan their budgets.

Style:
- Concise. Short sentences. No bullet lists unless asked.
- Use Korean if the user writes in Korean; otherwise English.
- Currency is CAD unless stated.

When you need data, call a tool. Don't make up transactions, balances, or categories.
When a tool returns { error: ... }, briefly explain the issue and suggest what to try.

Don't ask permission to look up data — just call the tool and show the result.`;

export type ToolCallRecord = {
  toolName: string;
  args: unknown;
  result: { output: unknown } | { error: string };
};

type TextBlock = { type: "text"; markdown: string };
type ToolBlock = { type: "tool"; toolName: string; args: unknown; result: unknown };
type Block = TextBlock | ToolBlock;

type GeminiContent = {
  role: "user" | "model";
  parts: Array<
    | { text: string }
    | { functionCall: { name: string; args: Record<string, unknown> } }
    | { functionResponse: { name: string; response: Record<string, unknown> } }
  >;
};

export function toGeminiHistory(
  messages: Array<{ role: string; blocks: unknown }>,
): GeminiContent[] {
  const out: GeminiContent[] = [];
  for (const m of messages) {
    const blocks = m.blocks as Block[];
    if (m.role === "user") {
      out.push({ role: "user", parts: [{ text: (blocks[0] as TextBlock).markdown }] });
      continue;
    }
    if (m.role === "assistant") {
      out.push({ role: "model", parts: [{ text: (blocks[0] as TextBlock).markdown }] });
      continue;
    }
    // role === "tool"
    const t = blocks[0] as ToolBlock;
    const isError =
      t.result && typeof t.result === "object" && "error" in (t.result as object);
    const response = isError
      ? (t.result as { error: string })
      : (t.result as { output: unknown });
    out.push({
      role: "model",
      parts: [{ functionCall: { name: t.toolName, args: (t.args as Record<string, unknown>) ?? {} } }],
    });
    out.push({
      role: "user",
      parts: [{ functionResponse: { name: t.toolName, response: response as Record<string, unknown> } }],
    });
  }
  // Gemini requires the first content to have role "user".
  while (out.length > 0 && out[0].role !== "user") {
    out.shift();
  }
  return out;
}

  env: Env,

  const MAX_TOOL_CALLS = env.CHAT_MAX_TOOL_CALLS_PER_TURN ?? 10;
  const MODEL = env.CHAT_MODEL ?? "gemini-2.5-flash-lite";

  const toolCtx: ToolContext = { env, workspaceId: thread.workspaceId, userId };

  const recent = await prisma.chatMessage.findMany({
    where: { threadId: thread.id },
    orderBy: { createdAt: "desc" },
    take: HISTORY_LIMIT,
  });
  const historyChrono = [...recent].reverse();

  const sysInstruction = `${SYSTEM_PROMPT}\n\n${await buildFinancialContext(thread.workspaceId)}`;
  const tools = [{ functionDeclarations: toGeminiFunctionDeclarations(REGISTRY) }];

  const ai = new GoogleGenAI({ apiKey: env.GEMINI_API_KEY });
  let contents = toGeminiHistory(historyChrono);
  if (contents.length === 0) {
    throw new Error("chat_history_empty");
  }
  const toolCalls: ToolCallRecord[] = [];
  let dispatched = 0;

  while (true) {
    const resp = await ai.models.generateContent({
      model: MODEL,
      contents,
      config: { systemInstruction: sysInstruction, tools },
    });

    const fcs: FunctionCall[] = resp.functionCalls ?? [];
    if (fcs.length === 0) {
      return { assistantText: resp.text ?? "", toolCalls };
    }

    if (dispatched + fcs.length > MAX_TOOL_CALLS) {
      console.warn("chat: max tool calls exceeded", {
        threadId: thread.id, dispatched, requested: fcs.length,
      });
      return {
        assistantText: "I couldn't complete that — try a simpler request.",
        toolCalls,
      };
    }

    const newContents: GeminiContent[] = [...contents];
    let dispatchedThisRound = 0;
    for (const fc of fcs) {
      if (!fc.name) {
        console.warn("chat: function call missing name", { threadId: thread.id });
        continue;
      }
      const fcName = fc.name;
      const fcArgs = (fc.args ?? {}) as Record<string, unknown>;

      const dispatch = await dispatchTool(toolCtx, fcName, fcArgs, REGISTRY);
      const responseObj = dispatch.ok
        ? { output: dispatch.result }
        : { error: dispatch.error };

      toolCalls.push({ toolName: fcName, args: fcArgs, result: responseObj });

      await prisma.chatMessage.create({
        data: {
          threadId: thread.id,
          role: "tool",
          blocks: [{ type: "tool", toolName: fcName, args: fcArgs, result: responseObj }],
        },
      });

      newContents.push(
        { role: "model", parts: [{ functionCall: { name: fcName, args: fcArgs } }] },
        { role: "user", parts: [{ functionResponse: { name: fcName, response: responseObj } }] },
      );
      dispatched++;
      dispatchedThisRound++;
    }

    if (dispatchedThisRound === 0) {
      console.warn("chat: function-call round with no dispatchable calls", {
        threadId: thread.id, requested: fcs.length,
      });
      return {
        assistantText: "I couldn't complete that — try a simpler request.",
        toolCalls,
      };
    }

    contents = newContents;
  }
}

import type { FastifyInstance } from "fastify";
import { z } from "zod";
import { prisma } from "../db.js";
import { HttpError, requireWorkspaceContext } from "../auth/context.js";
import { effectiveTier } from "../auth/tier.js";
import { processChatTurn } from "../chat/turn.js";

const Body = z.object({ text: z.string().min(1).max(4000) });

export async function chatRoutes(app: FastifyInstance): Promise<void> {
  app.post("/workspaces/:id/chat/message", async (req, reply) => {
    const params = req.params as { id: string };
    const parsed = Body.safeParse(req.body);
    if (!parsed.success) return reply.code(400).send({ error: "invalid_body" });

    try {
      const ctx = await requireWorkspaceContext(req, app, { requiredRole: "editor" });
      if (ctx.membership.workspaceId !== params.id) {
        return reply.code(400).send({ error: "workspace_mismatch" });
      }

      const tier = await effectiveTier(ctx.session.userId);
      if (tier.tier !== "pro") {
        return reply.code(402).send({ error: "upgrade_required" });
      }

      const thread = await prisma.chatThread.upsert({
        where: { workspaceId_type: { workspaceId: ctx.membership.workspaceId, type: "adhoc" } },
        create: { workspaceId: ctx.membership.workspaceId, type: "adhoc" },
        update: {},
      });

      const userMessage = await prisma.chatMessage.create({
        data: {
          threadId: thread.id,
          role: "user",
          blocks: [{ type: "text", markdown: parsed.data.text }],
        },
      });

      let turn: Awaited<ReturnType<typeof processChatTurn>>;
      try {
        turn = await processChatTurn(app.fina.env, { id: thread.id, workspaceId: thread.workspaceId }, ctx.session.userId);
      } catch (e) {
        console.error("chat provider failed", { threadId: thread.id, err: e });
        return reply.code(500).send({ error: "chat_provider_error" });
      }

      const assistantMessage = await prisma.chatMessage.create({
        data: {
          threadId: thread.id,
          role: "assistant",
          blocks: [{ type: "text", markdown: turn.assistantText }],
        },
      });

      return {
        threadId: thread.id,
        userMessage: {
          id: userMessage.id,
          role: userMessage.role,
          blocks: userMessage.blocks,
          createdAt: userMessage.createdAt.toISOString(),
        },
        assistantMessage: {
          id: assistantMessage.id,
          role: assistantMessage.role,
          blocks: assistantMessage.blocks,
          createdAt: assistantMessage.createdAt.toISOString(),
        },
        toolCalls: turn.toolCalls,
      };
    } catch (e) {
      if (e instanceof HttpError) return reply.code(e.statusCode).send({ error: e.code });
      throw e;
    }
  });

  app.get("/workspaces/:id/chat/thread", async (req, reply) => {
    const params = req.params as { id: string };
    try {
      const ctx = await requireWorkspaceContext(req, app, { requiredRole: "viewer" });
      if (ctx.membership.workspaceId !== params.id) {
        return reply.code(400).send({ error: "workspace_mismatch" });
      }

      const thread = await prisma.chatThread.upsert({
        where: { workspaceId_type: { workspaceId: ctx.membership.workspaceId, type: "adhoc" } },
        create: { workspaceId: ctx.membership.workspaceId, type: "adhoc" },
        update: {},
      });

      const messages = await prisma.chatMessage.findMany({
        where: { threadId: thread.id },
        orderBy: { createdAt: "asc" },
        take: 100,
      });

      return {
        threadId: thread.id,
        messages: messages.map((m) => ({
          id: m.id,
          role: m.role,
          blocks: m.blocks,
          createdAt: m.createdAt.toISOString(),
        })),
      };
    } catch (e) {
      if (e instanceof HttpError) return reply.code(e.statusCode).send({ error: e.code });
      throw e;
    }
  });
}
