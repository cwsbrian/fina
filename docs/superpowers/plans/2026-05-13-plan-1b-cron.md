# Plan 1B-cron — Scheduled background jobs

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add four in-process cron jobs (daily sync refresh, lock_held retry, event TTL cleanup, tier-gate enforcement) to the Fina API server.

**Architecture:** `croner` schedules three pure async job functions registered from `main.ts` when `CRON_ENABLED=true`. Jobs live in `apps/api/src/cron/jobs/`. A shared `tier-gate.ts` helper wraps the existing `effectiveTier()` for background-sync use. Tests invoke job functions directly (no cron timing involved).

**Tech Stack:** croner (TypeScript-native scheduler), Prisma (DB), Vitest (tests), existing `syncTransactions` / `dispatchWebhookSync` / `effectiveTier` from the codebase.

```ts
import { effectiveTier } from "../auth/tier.js";

export type GateResult = { ok: true } | { ok: false; reason: string };

/**
 * Background-sync version of requireProTier — never throws; returns a
 * structured skip reason so callers can log and continue.
 *
 * Reuses effectiveTier so trial/subscription logic stays in one place.
 * When paid subscriptions land (future plan), effectiveTier will return
 * tier:"pro" for subscribers — the first branch handles that correctly.
 */
export async function checkBackgroundSyncGate(userId: string): Promise<GateResult> {
  const t = await effectiveTier(userId);
  if (t.tier === "pro") return { ok: true };
  if (t.trialEndsAt && t.trialEndsAt <= new Date()) return { ok: false, reason: "trial_expired" };
  return { ok: false, reason: "no_subscription" };
}
```

```ts
import { prisma } from "../../db.js";
import type { Env } from "../../env.js";

export async function runCleanupEvents(env: Env): Promise<void> {
  const ttlDays = env.CRON_EVENT_TTL_DAYS ?? 30;
  const cutoff = new Date(Date.now() - ttlDays * 86_400_000);
  // Intentionally no processedAt filter: rows still NULL at TTL days are
  // superseded by cursor-driven syncs or permanently stuck. TTL sweeps them.
  const r = await prisma.plaidWebhookEvent.deleteMany({
    where: { receivedAt: { lt: cutoff } },
  });
  console.info("cleanup-events complete", { deleted: r.count, cutoff });
}
```

```ts
import { prisma } from "../../db.js";
import type { Env } from "../../env.js";
import { syncTransactions } from "../../plaid/sync.js";
import { categorizeNewTransactions } from "../../categorize/service.js";
import { sanitizePlaidError } from "../../plaid/client.js";
import { checkBackgroundSyncGate } from "../tier-gate.js";

export async function runDailyRefresh(env: Env): Promise<void> {
  const items = await prisma.plaidItem.findMany({
    where: { status: "active" },
    // connectedBy is the userId who ran Plaid Link — used for tier gating.
    // Workspace has no ownerUserId column. connectedBy is nullable (SetNull on user delete).
    select: { id: true, workspaceId: true, connectedBy: true },
  });

  const t0 = Date.now();
  let synced = 0, gated = 0, failed = 0, lockHeld = 0;

  for (const item of items) {
    if (item.connectedBy) {
      const gate = await checkBackgroundSyncGate(item.connectedBy);
      if (!gate.ok) { gated++; continue; }
    } else {
      console.warn("daily-refresh: connectedBy null, skipping tier gate", { itemId: item.id });
    }

    try {
      const r = await syncTransactions(env, item.workspaceId, item.id);
      if ("skipped" in r && r.skipped === "lock_held") {
        lockHeld++;
      } else {
        synced++;
      }
      if ("newTxIds" in r && r.newTxIds.length > 0) {
        await categorizeNewTransactions(env, item.workspaceId, r.newTxIds).catch((e) =>
          console.error("categorize after daily-refresh failed", { itemId: item.id, err: e }),
        );
      }
    } catch (e) {
      failed++;
      console.error("daily-refresh sync failed", { itemId: item.id, err: sanitizePlaidError(e) });
    }
  }

  console.info("daily-refresh complete", {
    durationMs: Date.now() - t0,
    itemsTotal: items.length, synced, gated, lockHeld, failed,
  });
}
```

```ts
import { prisma } from "../../db.js";
import type { Env } from "../../env.js";
import { dispatchWebhookSync } from "../../plaid/webhook-dispatch.js";
import { checkBackgroundSyncGate } from "../tier-gate.js";

export async function runRetryUnprocessed(env: Env): Promise<void> {
  const cutoff = new Date(Date.now() - 24 * 3600_000);

  const events = await prisma.plaidWebhookEvent.findMany({
    where: {
      processedAt: null,
      webhookCode: "SYNC_UPDATES_AVAILABLE",
      plaidItemId: { not: null },
      receivedAt: { gte: cutoff },
    },
    // No workspace join — tier gate uses plaidItem.connectedBy directly.
    include: { plaidItem: { select: { id: true, workspaceId: true, connectedBy: true } } },
    take: 50,
    orderBy: { receivedAt: "asc" },
  });

  let attempted = 0, gated = 0;

  for (const ev of events) {
    // Guard: plaidItemId filter ensures non-null at query time, but onDelete:SetNull
    // can race between findMany and here.
    if (!ev.plaidItem) continue;

    const item = ev.plaidItem;

    // Positive-condition form avoids an empty-if lint warning.
    if (item.connectedBy) {
      const gate = await checkBackgroundSyncGate(item.connectedBy);
      if (!gate.ok) {
        // Stamp the row so it stops appearing in the NULL retry set.
        await prisma.plaidWebhookEvent.update({
          where: { id: ev.id },
          data: { processedAt: new Date(), processError: `tier_gated:${gate.reason}` },
        });
        gated++;
        continue;
      }
    }
    // connectedBy null → pre-dates tracking; allow sync, no gate possible.

    // dispatchWebhookSync handles its own try/catch + processedAt semantics:
    //  - normal sync:  processedAt set
    //  - sync throws:  processedAt + processError set
    //  - lock_held:    processedAt stays NULL (row reappears next tick).
    //                  NOTE: any newTxIds from already-committed pages are still
    //                  categorized inside dispatchWebhookSync before it returns.
    await dispatchWebhookSync(env, item.workspaceId, item.id, ev.id);
    attempted++;
  }

  console.info("retry-unprocessed complete", { found: events.length, attempted, gated });
}
```

```ts
import { Cron } from "croner";
import type { Env } from "../env.js";
import { runDailyRefresh } from "./jobs/daily-refresh.js";
import { runRetryUnprocessed } from "./jobs/retry-unprocessed.js";
import { runCleanupEvents } from "./jobs/cleanup-events.js";

let jobs: Cron[] = [];

export function startScheduler(env: Env): void {
  if (!env.CRON_ENABLED) return;

  jobs = [
    new Cron("0 9 * * *",    { protect: true, name: "daily-refresh" },     () => safe("daily-refresh",     () => runDailyRefresh(env))),
    new Cron("*/10 * * * *", { protect: true, name: "retry-unprocessed" }, () => safe("retry-unprocessed", () => runRetryUnprocessed(env))),
    new Cron("0 3 * * *",    { protect: true, name: "cleanup-events" },    () => safe("cleanup-events",    () => runCleanupEvents(env))),
  ];

  console.info("cron scheduler started", { jobs: jobs.map((j) => j.name) });
}

export function stopScheduler(): void {
  for (const j of jobs) j.stop();
  jobs = [];
}

async function safe(name: string, fn: () => Promise<void>): Promise<void> {
  try {
    await fn();
  } catch (e) {
    console.error("cron job uncaught", { jobName: name, err: e });
  }
}
```
