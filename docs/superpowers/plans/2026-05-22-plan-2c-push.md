# Plan 2C-push Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deliver each 2B-generated daily brief to the user's registered mobile devices via Expo Push notifications (iOS + Android only; web push deferred).

**Architecture:** 10 task-commits, shared-first per CLAUDE.md. Backend gains a `UserDevice` table + POST /devices route + `expo-server-sdk`-backed `sendBriefPush` module + a brief-cron hook (post-`$transaction`). Mobile gains `expo-notifications` permission flow + Expo push token registration on launch and sign-in. Push send is fire-and-forget — brief persists in DB regardless.

**Tech Stack:** Fastify 5, Prisma 5, Pino 9, `expo-server-sdk` (new), `expo-notifications` + `expo-constants` (new on mobile), `@fina/shared`. Tests: Vitest singleFork, real Postgres at port 5433, mocked Expo SDKs.

```prisma
  id              String    @id @default(cuid())
  userId          String

  /// translates to APNs (iOS) or FCM (Android) device tokens internally.

  @@index([userId])
  @@map("user_devices")
```

Create `apps/api/src/routes/devices.ts`:

```ts
import type { FastifyInstance } from "fastify";
import { prisma } from "../db.js";
import { HttpError, requireSession } from "../auth/context.js";
import { RegisterDeviceBody } from "@fina/shared";

export async function devicesRoutes(app: FastifyInstance): Promise<void> {
  app.post("/devices", async (req, reply) => {
    try {
      const session = await requireSession(req, app);
      const parsed = RegisterDeviceBody.safeParse(req.body);
      if (!parsed.success) return reply.code(400).send({ error: "invalid_body" });
      const { expoPushToken, platform } = parsed.data;

      await prisma.userDevice.upsert({
        where: { userId_expoPushToken: { userId: session.userId, expoPushToken } },
        create: {
          userId: session.userId,
          expoPushToken,
          platform,
          lastSeenAt: new Date(),
        },
        update: {
          platform,           // in case user reinstalls; keep platform fresh
          lastSeenAt: new Date(),
        },
      });
      return reply.code(204).send();
    } catch (e) {
      if (e instanceof HttpError) return reply.code(e.statusCode).send({ error: e.code });
      throw e;
    }
  });
}
```

```ts
import { Expo, type ExpoPushMessage, type ExpoPushTicket } from "expo-server-sdk";
import { prisma } from "../db.js";
import { logInfo, logWarn, logError } from "../logger.js";
import type { Env } from "../env.js";

/** Carrier-message: ExpoPushMessage extended with our DB row id so the
 *  receipt-error handler knows which UserDevice to delete on
 *  DeviceNotRegistered. Expo's SDK ignores extra properties. */
type CarrierMessage = ExpoPushMessage & { __deviceId: string };

/** Result of one push send pass for a single user. */
export type PushSendResult = { sent: number; failed: number };

/** Send the daily brief preview to all of `userId`'s registered devices.
 *  Fire-and-forget — catches its own errors. Returns counts for logging.
 *
 *  Cleans up stale tokens inline: a DeviceNotRegistered ticket triggers
 *  deletion of the UserDevice row so the next send doesn't retry. */
export async function sendBriefPush(
  env: Env,
  userId: string,
  briefText: string,
): Promise<PushSendResult> {
  const devices = await prisma.userDevice.findMany({ where: { userId } });
  if (devices.length === 0) {
    logInfo({ userId, deviceCount: 0 }, "push send skipped — no devices");
    return { sent: 0, failed: 0 };
  }

  const expo = new Expo({ accessToken: env.EXPO_ACCESS_TOKEN });

  // Build carrier messages — filter out malformed tokens.
  const messages: CarrierMessage[] = [];
  for (const d of devices) {
    if (!Expo.isExpoPushToken(d.expoPushToken)) {
      logWarn({ userId, removedDeviceId: d.id }, "skipping malformed push token");
      continue;
    }
    messages.push({
      to: d.expoPushToken,
      title: "Fina ☀️",
      body: briefText,  // caller already truncated/cleaned via markdownToPlainPreview
      data: { type: "daily-brief" },
      sound: "default",
      priority: "high",
      __deviceId: d.id,
    });
  }

  if (messages.length === 0) {
    return { sent: 0, failed: 0 };
  }

  const chunks = expo.chunkPushNotifications(messages);
  let sent = 0;
  let failed = 0;

  for (const chunk of chunks) {
    let tickets: ExpoPushTicket[];
    try {
      tickets = await expo.sendPushNotificationsAsync(chunk);
    } catch (err) {
      // Network / Expo outage — log and treat the whole chunk as failed but
      // don't delete tokens (the issue is transient).
      logError({ userId, err, deviceCount: chunk.length }, "expo send threw");
      failed += chunk.length;
      continue;
    }

    for (let i = 0; i < tickets.length; i++) {
      const t = tickets[i]!;
      if (t.status === "ok") {
        sent++;
        continue;
      }
      failed++;
      const deviceId = (chunk[i] as CarrierMessage).__deviceId;
      const errorCode = t.details?.error;
      if (errorCode === "DeviceNotRegistered") {
        await prisma.userDevice.delete({ where: { id: deviceId } });
        logInfo({ userId, removedDeviceId: deviceId }, "stale device removed");
      } else {
        // Other errors (MessageRateExceeded, InvalidCredentials, ...) — log
        // and continue. Don't delete; transient issue.
        logWarn(
          { userId, removedDeviceId: deviceId, err: t.message },
          "push send error (token kept)",
        );
      }
    }
  }

  logInfo({ userId, deviceCount: devices.length, pushSent: sent, pushFailed: failed }, "push send done");
  return { sent, failed };
}
```

```ts
/** Strip markdown syntax + collapse newlines + truncate, for use as the
 *  body of a push notification where OS limits visible characters to
 *  ~120-180 depending on platform. Naive implementation — e.g., `5*3=15`
 *  becomes `53=15` — acceptable for brief content which is prose. */
export function markdownToPlainPreview(md: string, max = 120): string {
  const cleaned = md
    // Convert [text](url) → text BEFORE stripping link chars.
    .replace(/\[(.+?)\]\(.+?\)/g, "$1")
    // Strip common markdown markers.
    .replace(/[*_#`~>]/g, "")
    // Collapse all whitespace runs (incl. newlines) to a single space.
    .replace(/\s+/g, " ")
    .trim();

  if (cleaned.length <= max) return cleaned;
  return cleaned.slice(0, max) + "…";
}
```

```ts
import * as Notifications from "expo-notifications";
import Constants from "expo-constants";
import { Platform } from "react-native";
import { apiFetch } from "./client";

/** Best-effort registration of this device's Expo push token with the
 *  Fina API. Idempotent: safe to call on every launch + on sign-in.
 *  Returns the token on success, null on any failure (permission denied,
 *  missing app config, API down, etc.).
 *
 *  Failures are silent — push is a "nice to have" wake-up channel; the
 *  brief still appears in the chat thread next time the app foregrounds. */
export async function registerForPushNotifications(): Promise<string | null> {
  // OS permission (iOS dialog / Android-13+ runtime).
  const { status: existing } = await Notifications.getPermissionsAsync();
  let finalStatus = existing;
  if (existing !== "granted") {
    const { status } = await Notifications.requestPermissionsAsync();
    finalStatus = status;
  }
  if (finalStatus !== "granted") return null;

  // Read EAS project ID — required by Expo SDK 54's getExpoPushTokenAsync.
  const projectId =
    Constants.expoConfig?.extra?.eas?.projectId ??
    (Constants as { easConfig?: { projectId?: string } }).easConfig?.projectId;
  if (!projectId || projectId === "REPLACE_WITH_EAS_PROJECT_ID") {
    // EAS projectId not in app.json (or still the placeholder from T9). Silently
    // skip — add via `eas init` once the project is initialized on Expo's
    // dashboard. The placeholder check makes failure mode explicit instead of
    // relying on getExpoPushTokenAsync throwing.
    return null;
  }

  let token: string;
  try {
    const resp = await Notifications.getExpoPushTokenAsync({ projectId });
    token = resp.data;
  } catch {
    return null;
  }
  if (!token) return null;

  const platform = Platform.OS === "ios" ? "ios" : "android";
  try {
    await apiFetch("/devices", {
      method: "POST",
      body: JSON.stringify({ expoPushToken: token, platform }),
    });
  } catch {
    // Backend may be temporarily down. Token cached locally by Expo —
    // next launch will retry.
  }
  return token;
}
```

Create `apps/api/tests/push.integration.test.ts`:

```ts
import { describe, expect, it, beforeEach, afterEach, afterAll, vi } from "vitest";
import { prisma } from "../src/db.js";
import { buildApp } from "../src/server.js";
import { signUpAndGetCookie } from "./helpers/auth.js";
import { truncateAll, disconnect } from "./helpers/db.js";
import { testEnv } from "./helpers/test-env.js";

vi.mock("@google/genai", () => ({
  GoogleGenAI: vi.fn().mockImplementation(() => ({
    models: {
      generateContent: vi.fn().mockResolvedValue({
        text: "좋은 아침이야 ☀️ 어제 Starbucks 한 건 분류 필요해. 오늘 시장 $100 잡혀있어. 잘 보내!",
      }),
    },
  })),
}));

vi.mock("expo-server-sdk", () => ({
  Expo: class {
    static isExpoPushToken(t: string) { return /^ExponentPushToken\[/.test(t); }
    chunkPushNotifications(messages: unknown[]) { return [messages]; }
    async sendPushNotificationsAsync(_messages: unknown[]) {
      return [{ status: "ok", id: "t1" }];
    }
  },
}));

beforeEach(async () => { await truncateAll(); });
afterEach(() => { vi.restoreAllMocks(); });  // Vitest singleFork — prevent spy leakage across files.
afterAll(async () => { await disconnect(); });

describe("E2E: sign-up → register device → cron fires → push dispatched", () => {
  it("full chain", async () => {
    const app = await buildApp(testEnv());

    // 1) Sign up.
    const { cookie, userId } = await signUpAndGetCookie(app, { email: "e2e-push@t.co" });
    const ws = await prisma.workspace.create({ data: { name: "P" } });
    await prisma.membership.create({ data: { userId, workspaceId: ws.id, role: "owner" } });

    // 2) Register device via POST /devices.
    const registerRes = await app.inject({
      method: "POST",
      url: "/devices",
      headers: { cookie, "content-type": "application/json" },
      payload: JSON.stringify({ expoPushToken: "ExponentPushToken[e2e]", platform: "ios" }),
    });
    expect(registerRes.statusCode).toBe(204);

    // 3) Set morningTime to current minute in user's tz so cron picks them up.
    const partsInTz = (await import("../src/chat/tools/periods.js")).partsInTz;
    const tz = "Asia/Seoul";
    const pad2 = (n: number) => String(n).padStart(2, "0");
    const p = partsInTz(new Date(), tz);
    const nowHHmm = `${pad2(p.hour)}:${pad2(p.minute)}`;
    await prisma.user.update({ where: { id: userId }, data: { tz, morningTime: nowHHmm } });

    // Re-sync right before cron to handle minute rollover.
    const p2 = partsInTz(new Date(), tz);
    const nowHHmm2 = `${pad2(p2.hour)}:${pad2(p2.minute)}`;
    if (nowHHmm2 !== nowHHmm) {
      await prisma.user.update({ where: { id: userId }, data: { morningTime: nowHHmm2 } });
    }

    // 4) Fire cron.
    const { Expo } = await import("expo-server-sdk");
    const sendSpy = vi.spyOn(Expo.prototype, "sendPushNotificationsAsync");
    const { runDailyBrief } = await import("../src/cron/jobs/daily-brief.js");
    await runDailyBrief(testEnv());

    // 5) Verify brief in DB + push dispatched.
    const briefs = await prisma.brief.findMany({ where: { userId } });
    expect(briefs).toHaveLength(1);
    expect(sendSpy).toHaveBeenCalledTimes(1);
    const sendArg = sendSpy.mock.calls[0]?.[0] as Array<{ body: string; to: string }>;
    expect(sendArg[0]?.to).toBe("ExponentPushToken[e2e]");
    expect(sendArg[0]?.body).toMatch(/Starbucks/);

    await app.close();
  });
});
```
