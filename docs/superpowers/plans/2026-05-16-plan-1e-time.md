# Plan 1E-time Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fix user-timezone correctness in transaction APIs end-to-end: anchor Plaid date-only sync rows at UTC noon, backfill existing rows, and migrate the two remaining chat tools (`listTransactions`, `getSpendingSummary`) off server-local Date math onto the shared `periods.ts` helpers.

**Architecture:** One `parsePlaidInstant` helper in `apps/api/src/plaid/sync.ts` encapsulates the date-only → UTC noon policy; both `plaidOccurredAt` and `plaidPostedAt` route through it. A single Prisma migration shifts pre-existing date-only-origin Plaid rows by +12h, gated on `rawData` JSONB to avoid touching real ISO timestamps. `periods.ts` gains a `"today"` `BasicPeriod`. The two old tools call `basicPeriodRange(userId, period)` and `dateKeyInTz(occurredAt, tz)` so user-tz applies to both filter bounds and day-bucket keys.

**Tech Stack:** TypeScript ESM monorepo, Prisma + Postgres on 5433, Vitest (singleFork), Fastify.

**Spec:** [`docs/superpowers/specs/2026-05-16-plan-1e-time-design.md`](../specs/2026-05-16-plan-1e-time-design.md)

**Worktree:** This plan should be implemented in an isolated worktree branched from main. The subagent-driven-development skill will set this up via `EnterWorktree`.

```typescript
export function plaidOccurredAt(t: PlaidTxLike): Date {
  // Prefer authorized_* (when the user actually transacted), fall back to date.
  const v = t.authorized_datetime ?? t.authorized_date ?? t.datetime ?? t.date;
  if (!v) throw new Error(`Plaid transaction ${t.transaction_id} has no date field`);
  return parsePlaidInstant(v);
}

export function plaidPostedAt(t: PlaidTxLike): Date | null {
  if (t.pending) return null;
  const v = t.datetime ?? t.date;
  return parsePlaidInstant(v);
}
```

- [ ] **Step 2: Populate `migration.sql`**

Write to `apps/api/prisma/migrations/<ts>_plaid_date_to_utc_noon/migration.sql`:

```sql
-- Plan 1E-time: Plaid date-only transactions historically stored as UTC
-- midnight. Shift to UTC noon so user-tz day-boundary filters classify
-- correctly.
--
-- Gate logic mirrors the sync.ts fallback chain so real ISO Plaid timestamps
-- that happen to land at 00:00:00Z are NOT touched.
--   plaidOccurredAt prefers authorized_datetime > authorized_date > datetime > date
--   plaidPostedAt   prefers datetime > date
-- The trailing EXTRACT(HOUR/MINUTE/SECOND ...)=0 clauses make the migration
-- idempotent (a row already at 12:00 no longer matches).

UPDATE "transactions"
SET "occurredAt" = "occurredAt" + INTERVAL '12 hours'
WHERE "source" = 'plaid'
  AND ("rawData" ->> 'authorized_datetime') IS NULL
  AND (
    ("rawData" ->> 'authorized_date') IS NOT NULL
    OR (
          ("rawData" ->> 'authorized_date') IS NULL
      AND ("rawData" ->> 'datetime')        IS NULL
      AND ("rawData" ->> 'date')            IS NOT NULL
    )
  )
  AND EXTRACT(HOUR   FROM "occurredAt") = 0
  AND EXTRACT(MINUTE FROM "occurredAt") = 0
  AND EXTRACT(SECOND FROM "occurredAt") = 0;

UPDATE "transactions"
SET "postedAt" = "postedAt" + INTERVAL '12 hours'
WHERE "source" = 'plaid'
  AND "postedAt" IS NOT NULL
  AND ("rawData" ->> 'datetime') IS NULL
  AND ("rawData" ->> 'date')     IS NOT NULL
  AND EXTRACT(HOUR   FROM "postedAt") = 0
  AND EXTRACT(MINUTE FROM "postedAt") = 0
  AND EXTRACT(SECOND FROM "postedAt") = 0;
```

```typescript
export async function basicPeriodRange(
  userId: string,
  period: BasicPeriod,
  now: Date = new Date(),
): Promise<{ gte: Date; lt: Date }> {
  const tz = await resolveUserTz(userId);
  const p = partsInTz(now, tz);
  if (period === "today") {
    const gte = localMidnightToUtc(p.year, p.month, p.day, tz);
    return { gte, lt: addDaysLocal(gte, 1, tz) };
  }
  if (period === "this_month") {
    const gte = startOfMonth(p.year, p.month, tz);
    const next = nextMonth(p.year, p.month);
    return { gte, lt: startOfMonth(next.year, next.month, tz) };
  }
  if (period === "last_month") {
    const prev = prevMonth(p.year, p.month);
    return { gte: startOfMonth(prev.year, prev.month, tz), lt: startOfMonth(p.year, p.month, tz) };
  }
  // this_week
  const gte = startOfWeek(p.year, p.month, p.day, p.weekday, tz);
  return { gte, lt: addDaysLocal(gte, 7, tz) };
}
```

```typescript
import { basicPeriodRange } from "./periods.js";
import type { ToolDef } from "./types.js";

const Args = z.object({
  period: z.enum(["today", "this_week", "this_month", "last_month"]).describe("Time window for the listing. Omit for all-time.").optional(),
  limit: z.number().int().positive().max(50).describe("Max transactions to return. Default 20, hard cap 50.").optional(),

export const listTransactions: ToolDef<Args, { transactions: Array<{ id: string; amount: string; merchantRaw: string; merchantNormalized: string | null; categoryId: string | null; occurredAt: string }> }> = {
  name: "listTransactions",

  args: Args,
  async run(ctx, args) {
```

```typescript
  description: "Total spending in a period bucketed by category, merchant, or day.",
  args: Args,
  async run(ctx, args) {
    const groupBy = args.group_by ?? "category";
    const { gte, lt } = await basicPeriodRange(ctx.userId, args.period);
    const tz = await resolveUserTz(ctx.userId);
    const rows = await prisma.transaction.findMany({
      where: { workspaceId: ctx.workspaceId, removedAt: null, occurredAt: { gte, lt } },
      select: { amount: true, categoryId: true, merchantNormalized: true, occurredAt: true },
    });
    const buckets = new Map<string, { total: Prisma.Decimal; count: number }>();
    for (const r of rows) {
      let key: string;
      if (groupBy === "category") key = r.categoryId ?? "uncategorized";
      else if (groupBy === "merchant") key = r.merchantNormalized ?? "unknown";
      else key = dateKeyInTz(r.occurredAt, tz);
      const cur = buckets.get(key);
      if (cur) { cur.total = cur.total.plus(r.amount); cur.count += 1; }
      else buckets.set(key, { total: new Prisma.Decimal(r.amount), count: 1 });
    }
    const groups = [...buckets.entries()].map(([key, v]) => ({ key, total: v.total.toFixed(2), count: v.count })).sort((a, b) => parseFloat(b.total) - parseFloat(a.total));
    return { period: args.period, groupBy, groups };
  },
};
```
