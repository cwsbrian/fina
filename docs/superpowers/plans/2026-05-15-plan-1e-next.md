## Shared Conventions (apply to every task)

The engineer MUST follow these without re-confirmation. Each rule is grounded in the spec and the existing codebase.

1. **Imports.** Always import from `.js` extensions because the project compiles to ESM (`"../../db.js"`, `"./types.js"`).
2. **Prisma access.** Import `prisma` from `"../../db.js"`. Import `Prisma` from `"@prisma/client"` for `Prisma.Decimal`.
3. **Workspace isolation.** Every Prisma query MUST include `workspaceId: ctx.workspaceId`. Single-row lookups MUST use `findFirst({ where: { id, workspaceId } })`, never `findUnique({ where: { id } })`.
4. **Soft delete.** Aggregates and list-style tools filter `removedAt: null`. The detail tool returns soft-deleted rows when explicitly requested by id.
5. **Sign convention.** Plaid spending is stored with positive amounts. `amount > 0` = expense, `amount < 0` = income/refund. Spending aggregates filter `amount: { gt: 0 }`. Cashflow sums positives as expense and absolute negatives as income.
6. **Currency scope.** Aggregate tools filter `currency: "CAD"` and expose `excludedNonCadCount` for the same period.
7. **Decimal/date serialization.** `Prisma.Decimal` → `.toFixed(2)` for monetary totals (matching `get-spending-summary.ts`), `.toString()` for raw transaction amounts. Dates → `.toISOString()`. Date-only labels → `YYYY-MM-DD`.
8. **`.describe()` on every top-level Zod arg field.** Verified by the registry/schema test in Task 9.
9. **Errors.** Throw plain `new Error("snake_case_code")` (e.g., `"transaction_not_found"`, `"invalid_period_pair"`).
10. **Test framework.** Vitest. Tests live in `apps/api/tests/`. `prisma` is imported from `"../src/db.js"`. `ToolContext` is imported from `"../src/chat/tools/types.js"`. Use `beforeEach` to seed and `afterEach` to delete the workspace.
11. **Test ctx shape.** `{ env: {} as never, workspaceId, userId }`.
12. **Commands.**
    - Run a single test file: `pnpm --filter @fina/api exec vitest run --config vitest.config.ts apps/api/tests/<file>.test.ts`
    - Run all api tests: `pnpm --filter @fina/api test`
    - Typecheck: `pnpm --filter @fina/api exec tsc --noEmit -p apps/api/tsconfig.app.json`
    - Test DB: `DATABASE_URL=postgresql://fina:fina@localhost:5433/fina_test` is configured via `.env.test`. The global setup applies migrations; the engineer does not need to run them manually.

## Task 1: Shared timezone-aware period helper (`periods.ts`)

**Files:**
- Create: `apps/api/src/chat/tools/periods.ts`
- Test: `apps/api/tests/chat.tools.periods.test.ts`

**Background:** Existing tools use server-local `Date` math via private `periodRange()` helpers. Plan 1E-next requires user-timezone-aware boundaries derived from `User.tz` (defaults to `"America/Toronto"` per the Prisma schema). We will not add a timezone library; we use `Intl.DateTimeFormat` with the user's timezone to derive Y/M/D fields and construct UTC `Date` instants from them.

**Period semantics:**
- Week boundary: Monday 00:00 local → next Monday 00:00 local (week starts Monday; matches existing `get-spending-summary.ts` behavior).
- Month boundary: 1st of month 00:00 local → 1st of next month 00:00 local.
- `previous_week` = week before `this_week`.
- `last_month` = calendar month before `this_month`.
- `all_time` = `{}` (no `gte`/`lt`).

### Step 1.3: Write the implementation

Create `apps/api/src/chat/tools/periods.ts`:

```typescript
import { prisma } from "../../db.js";

export type BasicPeriod = "this_week" | "this_month" | "last_month";
export type ReviewPeriod = BasicPeriod | "all_time";
export type CurrentPeriod = "this_week" | "this_month";
export type ComparisonPeriod = "previous_week" | "last_month";

const DEFAULT_TZ = "America/Toronto";

async function resolveTz(userId: string): Promise<string> {
  const user = await prisma.user.findUnique({ where: { id: userId }, select: { tz: true } });
  const tz = user?.tz ?? DEFAULT_TZ;
  if (!isValidTz(tz)) return DEFAULT_TZ;
  return tz;
}

function isValidTz(tz: string): boolean {
  try {
    new Intl.DateTimeFormat("en-US", { timeZone: tz });
    return true;
  } catch {
    return false;
  }
}

type Parts = { year: number; month: number; day: number; weekday: number };

function partsInTz(instant: Date, tz: string): Parts {
  const fmt = new Intl.DateTimeFormat("en-US", {
    timeZone: tz,
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
    weekday: "short",
    hour12: false,
  });
  const out: Record<string, string> = {};
  for (const p of fmt.formatToParts(instant)) {
    if (p.type !== "literal") out[p.type] = p.value;
  }
  const weekdayMap: Record<string, number> = { Sun: 0, Mon: 1, Tue: 2, Wed: 3, Thu: 4, Fri: 5, Sat: 6 };
  return {
    year: Number(out.year),
    month: Number(out.month),
    day: Number(out.day),
    weekday: weekdayMap[out.weekday] ?? 0,
  };
}

// Compute the UTC instant corresponding to (year, month, day, 00:00:00) interpreted in `tz`.
// Uses an offset-iteration trick: start with a UTC guess, measure the difference between
// (year, month, day) in `tz` and the target, and shift the UTC time by that difference.
// Converges in at most 2 iterations for any IANA zone.
function localMidnightToUtc(year: number, month: number, day: number, tz: string): Date {
  let guess = Date.UTC(year, month - 1, day, 0, 0, 0);
  for (let i = 0; i < 3; i++) {
    const got = partsInTz(new Date(guess), tz);
    const wantMs = Date.UTC(year, month - 1, day, 0, 0, 0);
    const gotMs = Date.UTC(got.year, got.month - 1, got.day, 0, 0, 0);
    const delta = wantMs - gotMs;
    if (delta === 0) return new Date(guess);
    guess += delta;
  }
  return new Date(guess);
}

function startOfMonth(year: number, month: number, tz: string): Date {
  return localMidnightToUtc(year, month, 1, tz);
}

function nextMonth(year: number, month: number): { year: number; month: number } {
  return month === 12 ? { year: year + 1, month: 1 } : { year, month: month + 1 };
}

function prevMonth(year: number, month: number): { year: number; month: number } {
  return month === 1 ? { year: year - 1, month: 12 } : { year, month: month - 1 };
}

function startOfWeek(year: number, month: number, day: number, weekday: number, tz: string): Date {
  // Monday-start: distance back from current local weekday to Monday.
  const back = (weekday + 6) % 7; // Sun=0 → 6, Mon=1 → 0, ...
  const noon = localMidnightToUtc(year, month, day, tz);
  // Subtract `back` whole days. Because tz-local midnights are 24h apart in IANA zones except across DST transitions,
  // we round-trip through localMidnightToUtc again.
  const shifted = new Date(noon.getTime() - back * 24 * 60 * 60 * 1000);
  const p = partsInTz(shifted, tz);
  return localMidnightToUtc(p.year, p.month, p.day, tz);
}

function addDays(d: Date, n: number, tz: string): Date {
  const shifted = new Date(d.getTime() + n * 24 * 60 * 60 * 1000);
  const p = partsInTz(shifted, tz);
  return localMidnightToUtc(p.year, p.month, p.day, tz);
}

export async function basicPeriodRange(
  userId: string,
  period: BasicPeriod,
  now: Date = new Date(),
): Promise<{ gte: Date; lt: Date }> {
  const tz = await resolveTz(userId);
  const p = partsInTz(now, tz);
  if (period === "this_month") {
    const gte = startOfMonth(p.year, p.month, tz);
    const next = nextMonth(p.year, p.month);
    return { gte, lt: startOfMonth(next.year, next.month, tz) };
  }
  if (period === "last_month") {
    const prev = prevMonth(p.year, p.month);
    return { gte: startOfMonth(prev.year, prev.month, tz), lt: startOfMonth(p.year, p.month, tz) };
  }
  // this_week
  const gte = startOfWeek(p.year, p.month, p.day, p.weekday, tz);
  return { gte, lt: addDays(gte, 7, tz) };
}

export async function reviewPeriodRange(
  userId: string,
  period: ReviewPeriod,
  now: Date = new Date(),
): Promise<{ gte?: Date; lt?: Date }> {
  if (period === "all_time") return {};
  return basicPeriodRange(userId, period, now);
}

export async function comparisonRanges(
  userId: string,
  currentPeriod: CurrentPeriod,
  comparisonPeriod: ComparisonPeriod,
  now: Date = new Date(),
): Promise<{ current: { gte: Date; lt: Date }; comparison: { gte: Date; lt: Date } }> {
  if (currentPeriod === "this_month" && comparisonPeriod === "last_month") {
    const current = await basicPeriodRange(userId, "this_month", now);
    const comparison = await basicPeriodRange(userId, "last_month", now);
    return { current, comparison };
  }
  if (currentPeriod === "this_week" && comparisonPeriod === "previous_week") {
    const current = await basicPeriodRange(userId, "this_week", now);
    const tz = await resolveTz(userId);
    const comparisonGte = addDays(current.gte, -7, tz);
    return { current, comparison: { gte: comparisonGte, lt: current.gte } };
  }
  throw new Error("invalid_period_pair");
}
```

> **Plaid date-only caveat:** Plaid sync currently stores date-only transactions as UTC-midnight instants. User-timezone period filters can therefore mis-bin transactions at the exact day boundary for non-UTC zones. This plan does not change storage. A future plan (1E-time in the spec) will address it.

```typescript
import { z } from "zod";
import { prisma } from "../../db.js";
import type { ToolDef } from "./types.js";

const Args = z.object({
  include_archived: z
    .boolean()
    .describe("Whether to include archived bank accounts. Defaults to false.")
    .optional(),
});

type Args = z.infer<typeof Args>;

type Account = {
  id: string;
  name: string;
  officialName: string | null;
  type: string;
  subtype: string | null;
  currency: string;
  status: "active" | "closed" | "hidden";
  balanceCurrent: string | null;
  balanceAvailable: string | null;
  balanceLimit: string | null;
  lastBalanceSyncAt: string | null;
  archivedAt: string | null;
  institutionName: string | null;
  plaidItemStatus: "active" | "error" | "relink_required" | "disconnected" | null;
  plaidItemErrorCode: string | null;
  plaidItemLastSyncAt: string | null;
};

export const listAccounts: ToolDef<Args, { accounts: Account[] }> = {
  name: "listAccounts",
  description:
    "List bank accounts in the workspace with balances, account status, and Plaid connection status.",
  args: Args,
  async run(ctx, args) {
    const where: { workspaceId: string; archivedAt?: null } = { workspaceId: ctx.workspaceId };
    if (args.include_archived !== true) where.archivedAt = null;

    const rows = await prisma.bankAccount.findMany({
      where,
      include: {
        plaidItem: {
          select: { institutionName: true, status: true, errorCode: true, lastSyncAt: true },
        },
      },
    });

    const statusOrder: Record<string, number> = { active: 0, hidden: 1, closed: 2 };
    rows.sort((a, b) => {
      const aArch = a.archivedAt ? 1 : 0;
      const bArch = b.archivedAt ? 1 : 0;
      if (aArch !== bArch) return aArch - bArch;
      const sa = statusOrder[a.status] ?? 99;
      const sb = statusOrder[b.status] ?? 99;
      if (sa !== sb) return sa - sb;
      if (a.name !== b.name) return a.name.localeCompare(b.name);
      return a.id.localeCompare(b.id);
    });

    const accounts: Account[] = rows.map((r) => ({
      id: r.id,
      name: r.name,
      officialName: r.officialName,
      type: r.type,
      subtype: r.subtype,
      currency: r.currency,
      status: r.status as Account["status"],
      balanceCurrent: r.balanceCurrent?.toFixed(2) ?? null,
      balanceAvailable: r.balanceAvailable?.toFixed(2) ?? null,
      balanceLimit: r.balanceLimit?.toFixed(2) ?? null,
      lastBalanceSyncAt: r.lastBalanceSyncAt?.toISOString() ?? null,
      archivedAt: r.archivedAt?.toISOString() ?? null,
      institutionName: r.plaidItem?.institutionName ?? null,
      plaidItemStatus: (r.plaidItem?.status ?? null) as Account["plaidItemStatus"],
      plaidItemErrorCode: r.plaidItem?.errorCode ?? null,
      plaidItemLastSyncAt: r.plaidItem?.lastSyncAt?.toISOString() ?? null,
    }));

    return { accounts };
  },
};
```

```typescript
import type { Prisma } from "@prisma/client";
import { prisma } from "../../db.js";
import type { ToolDef } from "./types.js";
import { reviewPeriodRange, type ReviewPeriod } from "./periods.js";

const Args = z.object({
  period: z
    .enum(["this_week", "this_month", "last_month", "all_time"])
    .describe("Time window to search for uncategorized transactions. Defaults to this_month.")
    .optional(),
  review_mode: z
    .enum(["missing_category", "not_manual"])
    .describe(
      "missing_category returns categoryId=null. not_manual returns transactions whose categorizationSource is not manual, including uncategorized.",
    )
    .optional(),
  limit: z
    .number()
    .int()
    .positive()
    .max(50)
    .describe("Maximum transactions to return. Defaults to 20, hard cap 50.")
    .optional(),
});

type Args = z.infer<typeof Args>;

type Result = {
  period: ReviewPeriod;
  reviewMode: "missing_category" | "not_manual";
  transactions: Array<{
    id: string;
    amount: string;
    currency: string;
    merchantRaw: string;
    merchantNormalized: string | null;
    occurredAt: string;
    accountName: string;
    categoryId: string | null;
    categorizationSource: "mapping" | "ai" | "manual" | null;
  }>;
};

export const findUncategorizedTransactions: ToolDef<Args, Result> = {
  name: "findUncategorizedTransactions",
  description: "Find transactions that are uncategorized or have not been manually reviewed.",
  args: Args,
  async run(ctx, args) {
    const period: ReviewPeriod = args.period ?? "this_month";
    const reviewMode = args.review_mode ?? "missing_category";
    const limit = args.limit ?? 20;
    const range = await reviewPeriodRange(ctx.userId, period);

    const where: Prisma.TransactionWhereInput = {
      workspaceId: ctx.workspaceId,
      removedAt: null,
    };
    if (range.gte !== undefined && range.lt !== undefined) {
      where.occurredAt = { gte: range.gte, lt: range.lt };
    }
    if (reviewMode === "missing_category") {
      where.categoryId = null;
    } else {
      where.OR = [{ categorizationSource: null }, { categorizationSource: { not: "manual" } }];
    }

    const rows = await prisma.transaction.findMany({
      where,
      orderBy: [{ occurredAt: "desc" }, { id: "desc" }],
      take: limit,
      include: { account: { select: { name: true } } },
    });

    return {
      period,
      reviewMode,
      transactions: rows.map((r) => ({
        id: r.id,
        amount: r.amount.toString(),
        currency: r.currency,
        merchantRaw: r.merchantRaw,
        merchantNormalized: r.merchantNormalized,
        occurredAt: r.occurredAt.toISOString(),
        accountName: r.account.name,
        categoryId: r.categoryId,
        categorizationSource: (r.categorizationSource ?? null) as Result["transactions"][number]["categorizationSource"],
      })),
    };
  },
};
```

```typescript
import { z } from "zod";
import { Prisma } from "@prisma/client";
import { prisma } from "../../db.js";
import type { ToolDef } from "./types.js";
import { basicPeriodRange, type BasicPeriod } from "./periods.js";

const Args = z.object({
  period: z
    .enum(["this_week", "this_month", "last_month"])
    .describe("Period to summarize spending by category."),
  include_uncategorized: z
    .boolean()
    .describe("Whether to include uncategorized spending as its own bucket. Defaults to true.")
    .optional(),
  limit: z
    .number()
    .int()
    .positive()
    .max(25)
    .describe("Maximum category buckets to return. Defaults to 10, hard cap 25.")
    .optional(),
});

type Args = z.infer<typeof Args>;

type Bucket = {
  categoryId: string | null;
  categoryName: string;
  parentCategoryId: string | null;
  parentCategoryName: string | null;
  color: string | null;
  icon: string | null;
  total: string;
  count: number;
  share: number;
};

type Result = {
  period: BasicPeriod;
  currency: "CAD";
  total: string;
  excludedNonCadCount: number;
  categories: Bucket[];
};

export const getCategoryBreakdown: ToolDef<Args, Result> = {
  name: "getCategoryBreakdown",
  description:
    "Summarize CAD spending by readable category name, including parent category and visual metadata.",
  args: Args,
  async run(ctx, args) {
    const includeUncat = args.include_uncategorized ?? true;
    const limit = args.limit ?? 10;
    const { gte, lt } = await basicPeriodRange(ctx.userId, args.period);

    const [rows, excludedNonCadCount] = await Promise.all([
      prisma.transaction.findMany({
        where: {
          workspaceId: ctx.workspaceId,
          removedAt: null,
          currency: "CAD",
          amount: { gt: 0 },
          occurredAt: { gte, lt },
        },
        select: {
          amount: true,
          categoryId: true,
          category: {
            select: {
              id: true,
              name: true,
              color: true,
              icon: true,
              parentId: true,
              parent: { select: { id: true, name: true } },
            },
          },
        },
      }),
      prisma.transaction.count({
        where: {
          workspaceId: ctx.workspaceId,
          removedAt: null,
          currency: { not: "CAD" },
          amount: { gt: 0 },
          occurredAt: { gte, lt },
        },
      }),
    ]);

    const buckets = new Map<string, { meta: Omit<Bucket, "total" | "count" | "share">; total: Prisma.Decimal; count: number }>();
    let total = new Prisma.Decimal(0);
    for (const r of rows) {
      const key = r.categoryId ?? "__uncat__";
      const meta: Omit<Bucket, "total" | "count" | "share"> = r.category
        ? {
            categoryId: r.category.id,
            categoryName: r.category.name,
            parentCategoryId: r.category.parent?.id ?? null,
            parentCategoryName: r.category.parent?.name ?? null,
            color: r.category.color,
            icon: r.category.icon,
          }
        : {
            categoryId: null,
            categoryName: "Uncategorized",
            parentCategoryId: null,
            parentCategoryName: null,
            color: null,
            icon: null,
          };
      const cur = buckets.get(key);
      if (cur) {
        cur.total = cur.total.plus(r.amount);
        cur.count += 1;
      } else {
        buckets.set(key, { meta, total: new Prisma.Decimal(r.amount), count: 1 });
      }
      total = total.plus(r.amount);
    }

    let entries = [...buckets.values()];
    if (!includeUncat) entries = entries.filter((e) => e.meta.categoryId !== null);

    const totalNum = parseFloat(total.toFixed(2));
    entries.sort((a, b) => {
      const diff = parseFloat(b.total.toFixed(2)) - parseFloat(a.total.toFixed(2));
      if (diff !== 0) return diff;
      return a.meta.categoryName.localeCompare(b.meta.categoryName);
    });

    const categories: Bucket[] = entries.slice(0, limit).map((e) => ({
      ...e.meta,
      total: e.total.toFixed(2),
      count: e.count,
      share: totalNum > 0 ? parseFloat(e.total.toFixed(2)) / totalNum : 0,
    }));

    return {
      period: args.period,
      currency: "CAD",
      total: total.toFixed(2),
```

```typescript
  comparison_period: z
    .enum(["previous_week", "last_month"])
    .describe("Comparison period. Use previous_week with this_week, last_month with this_month."),
  group_by: z.enum(["category", "merchant"]).describe("Dimension to compare spending by."),

    .number()
    .int()
    .positive()
    .max(25)
    .describe("Maximum changed buckets to return. Defaults to 10, hard cap 25.")
    .optional(),

type Change = {
  key: string;
  label: string;
  currentTotal: string;
  comparisonTotal: string;
  delta: string;
  percentChange: number | null;
  currentCount: number;
  comparisonCount: number;
};

type Result = {
  currentPeriod: CurrentPeriod;
  comparisonPeriod: ComparisonPeriod;
  groupBy: "category" | "merchant";
  currency: "CAD";
  currentTotal: string;
  comparisonTotal: string;
  delta: string;
  percentChange: number | null;
  currentExcludedNonCadCount: number;
  comparisonExcludedNonCadCount: number;
  changes: Change[];
};

function isValidPair(c: CurrentPeriod, p: ComparisonPeriod): boolean {
  return (c === "this_week" && p === "previous_week") || (c === "this_month" && p === "last_month");
}

type Row = {
  amount: Prisma.Decimal;
  categoryId: string | null;
  category: { id: string; name: string } | null;
  merchantRaw: string;
  merchantNormalized: string | null;
  occurredAt: Date;
};

type Bucket = { key: string; label: string; total: Prisma.Decimal; count: number };

function bucketize(rows: Row[], groupBy: "category" | "merchant"): {
  map: Map<string, Bucket>;
  total: Prisma.Decimal;
} {
  const map = new Map<string, Bucket>();
  let total = new Prisma.Decimal(0);
  for (const r of rows) {
    let key: string;
    let label: string;
    if (groupBy === "category") {
      key = r.categoryId ?? "uncategorized";
      label = r.category?.name ?? "Uncategorized";
    } else {
      const norm = r.merchantNormalized ?? normalizeMerchant(r.merchantRaw);
      key = norm;
      label = r.merchantRaw;
    }
    const cur = map.get(key);
    if (cur) {
      cur.total = cur.total.plus(r.amount);
      cur.count += 1;
      // prefer the most recent raw merchant as label for merchant mode
      if (groupBy === "merchant" && r.occurredAt.getTime() >= (cur as Bucket & { _ts?: number })._ts!) {
        cur.label = r.merchantRaw;
        (cur as Bucket & { _ts?: number })._ts = r.occurredAt.getTime();
      }
    } else {
      const b: Bucket & { _ts?: number } = { key, label, total: new Prisma.Decimal(r.amount), count: 1 };
      if (groupBy === "merchant") b._ts = r.occurredAt.getTime();
      map.set(key, b);
    }
    total = total.plus(r.amount);
  }
  return { map, total };
}

export const compareSpendingPeriods: ToolDef<Args, Result> = {
  name: "compareSpendingPeriods",
  description:
    "Compare CAD spending between two matching periods by category or merchant and return the largest changes.",
  args: Args,
  async run(ctx, args) {
    if (!isValidPair(args.current_period, args.comparison_period)) {
      throw new Error("invalid_period_pair");
    }
    const limit = args.limit ?? 10;
    const ranges = await comparisonRanges(ctx.userId, args.current_period, args.comparison_period);

    const selectSpec = {
      amount: true,
      categoryId: true,
      category: { select: { id: true, name: true } },
      merchantRaw: true,
      merchantNormalized: true,
      occurredAt: true,
    };

      prisma.transaction.findMany({
        where: {
          workspaceId: ctx.workspaceId,
          removedAt: null,
          currency: "CAD",
          amount: { gt: 0 },
          occurredAt: { gte: ranges.current.gte, lt: ranges.current.lt },
        },
        select: selectSpec,
      }),
      prisma.transaction.findMany({
        where: {
          workspaceId: ctx.workspaceId,
          removedAt: null,
          currency: "CAD",
          amount: { gt: 0 },
          occurredAt: { gte: ranges.comparison.gte, lt: ranges.comparison.lt },
        },
        select: selectSpec,
      }),
      prisma.transaction.count({
        where: {
          workspaceId: ctx.workspaceId,
          removedAt: null,
          currency: { not: "CAD" },
          amount: { gt: 0 },
          occurredAt: { gte: ranges.current.gte, lt: ranges.current.lt },
        },
      }),
      prisma.transaction.count({
        where: {
          workspaceId: ctx.workspaceId,
          removedAt: null,
          currency: { not: "CAD" },
          amount: { gt: 0 },
          occurredAt: { gte: ranges.comparison.gte, lt: ranges.comparison.lt },
        },
      }),
    ]);

    const current = bucketize(currentRows as Row[], args.group_by);
    const comparison = bucketize(comparisonRows as Row[], args.group_by);

    const allKeys = new Set<string>([...current.map.keys(), ...comparison.map.keys()]);
    const changes: Change[] = [];
    for (const key of allKeys) {
      const c = current.map.get(key);
      const p = comparison.map.get(key);
      const cTotal = c?.total ?? new Prisma.Decimal(0);
      const pTotal = p?.total ?? new Prisma.Decimal(0);
      const delta = cTotal.minus(pTotal);
      const pTotalNum = parseFloat(pTotal.toFixed(2));
      const percentChange = pTotalNum === 0 ? null : parseFloat(delta.toFixed(2)) / pTotalNum;
      const label = c?.label ?? p?.label ?? key;
      changes.push({
        key,
        label,
        currentTotal: cTotal.toFixed(2),
        comparisonTotal: pTotal.toFixed(2),
        delta: delta.toFixed(2),
        percentChange,
        currentCount: c?.count ?? 0,
        comparisonCount: p?.count ?? 0,
      });
    }
    changes.sort((a, b) => {
      const ad = Math.abs(parseFloat(a.delta));
      const bd = Math.abs(parseFloat(b.delta));
      if (bd !== ad) return bd - ad;
      return a.label.localeCompare(b.label);
    });

    const overallDelta = current.total.minus(comparison.total);
    const compTotalNum = parseFloat(comparison.total.toFixed(2));
    const overallPercent = compTotalNum === 0 ? null : parseFloat(overallDelta.toFixed(2)) / compTotalNum;

    return {
      currentPeriod: args.current_period,
      comparisonPeriod: args.comparison_period,
      groupBy: args.group_by,
      currency: "CAD",
      currentTotal: current.total.toFixed(2),
      comparisonTotal: comparison.total.toFixed(2),
      delta: overallDelta.toFixed(2),
      percentChange: overallPercent,
      currentExcludedNonCadCount: currentExcluded,
      comparisonExcludedNonCadCount: comparisonExcluded,
      changes: changes.slice(0, limit),
    };
  },
};
```

```typescript
    .describe("Maximum recurring merchant candidates to return. Defaults to 10, hard cap 25.")

type Candidate = {
  key: string;
  merchantNormalized: string;
  displayName: string;
  occurrences: number;
  firstSeenAt: string;
  lastSeenAt: string;
  medianAmount: string;
  latestAmount: string;
  currency: string;
  cadence: "weekly" | "biweekly" | "monthly" | "irregular";
  confidence: number;
  transactionIds: string[];
};

type Result = {
  lookbackMonths: number;
  candidates: Candidate[];
};

function median(nums: number[]): number {
  if (nums.length === 0) return 0;
  const s = [...nums].sort((a, b) => a - b);
  const mid = Math.floor(s.length / 2);
  return s.length % 2 === 0 ? (s[mid - 1] + s[mid]) / 2 : s[mid];
}

function classifyCadence(medianGapDays: number): Candidate["cadence"] {
  if (medianGapDays >= 6 && medianGapDays <= 8) return "weekly";
  if (medianGapDays >= 12 && medianGapDays <= 16) return "biweekly";
  if (medianGapDays >= 27 && medianGapDays <= 34) return "monthly";
  return "irregular";
}

function cadenceRegularityScore(gaps: number[], medianGap: number, cadence: Candidate["cadence"]): number {
  if (cadence === "irregular" || gaps.length === 0) return 0;
  // Mean absolute deviation as a fraction of medianGap. Lower MAD → higher score.
  const mad = gaps.reduce((s, g) => s + Math.abs(g - medianGap), 0) / gaps.length;
  const relative = mad / medianGap;
  return Math.max(0, 1 - Math.min(1, relative * 2)); // 0..1
}

function amountStabilityScore(amounts: number[]): number {
  if (amounts.length === 0) return 0;
  const med = median(amounts);
  if (med === 0) return 0;
  const mad = amounts.reduce((s, a) => s + Math.abs(a - med), 0) / amounts.length;
  return Math.max(0, 1 - Math.min(1, mad / med));
}

    // resolve user tz and use Intl-derived parts.

    const tz = user?.tz ?? "America/Toronto";

    const fmt = new Intl.DateTimeFormat("en-US", { timeZone: tz, year: "numeric", month: "2-digit" });
    const parts: Record<string, string> = {};
    for (const p of fmt.formatToParts(now)) if (p.type !== "literal") parts[p.type] = p.value;
    const baseYear = Number(parts.year);
    const baseMonth = Number(parts.month);
    let y = baseYear;
    let m = baseMonth - lookbackMonths + 1;
    while (m <= 0) {
      m += 12;
      y -= 1;
    }
    // first of (y, m) at UTC midnight is good enough as a lower bound
    const lookbackStart = new Date(Date.UTC(y, m - 1, 1));

        removedAt: null,

        amount: true,
        currency: true,
        merchantRaw: true,
        merchantNormalized: true,
        occurredAt: true,

      key: string;
      merchantKey: string;
      currency: string;
      displayName: string;

      if (!merchantKey) continue;

          currency: r.currency,
          displayName: r.merchantRaw,
          latestDisplayTs: r.occurredAt.getTime(),
          transactionIds: [],
          amounts: [],
          occurredAts: [],
        };
        groups.set(key, g);
      }
      g.transactionIds.push(r.id);
      g.amounts.push(new Prisma.Decimal(r.amount));
      g.occurredAts.push(r.occurredAt);
      if (r.occurredAt.getTime() >= g.latestDisplayTs) {
        g.displayName = r.merchantRaw;
        g.latestDisplayTs = r.occurredAt.getTime();
      }
    }

    const candidates: Candidate[] = [];
    for (const g of groups.values()) {
      if (g.occurredAts.length < minOccurrences) continue;
      const sortedTimes = g.occurredAts.map((d) => d.getTime()).sort((a, b) => a - b);
      const gaps: number[] = [];
      for (let i = 1; i < sortedTimes.length; i++) {
        gaps.push((sortedTimes[i] - sortedTimes[i - 1]) / (1000 * 60 * 60 * 24));
      }
      const medianGap = median(gaps);
      const cadence = classifyCadence(medianGap);
      const amountNums = g.amounts.map((a) => parseFloat(a.toFixed(2)));
      const medianAmt = median(amountNums);

      const occScore = Math.min(1, g.occurredAts.length / Math.max(minOccurrences, 4)) * 0.4;
      const cadenceScore = cadenceRegularityScore(gaps, medianGap, cadence) * 0.4;
      const stabilityScore = amountStabilityScore(amountNums) * 0.2;
      const confidence = Math.max(0, Math.min(1, occScore + cadenceScore + stabilityScore));

      if (confidence < 0.45) continue;

      // Find the amount for the latest occurrence by scanning the parallel arrays;
      // findIndex would return only the first match if timestamps collided.
      let latestAmount = g.amounts[0];
      let latestTs = g.occurredAts[0].getTime();
      for (let i = 1; i < g.amounts.length; i++) {
        const ts = g.occurredAts[i].getTime();
        if (ts > latestTs) {
          latestTs = ts;
          latestAmount = g.amounts[i];
        }
      }
      const firstTs = sortedTimes[0];
      const lastTs = sortedTimes[sortedTimes.length - 1];

      candidates.push({
        key: g.key,
        merchantNormalized: g.merchantKey,
        displayName: g.displayName,
        occurrences: g.occurredAts.length,
        firstSeenAt: new Date(firstTs).toISOString(),
        lastSeenAt: new Date(lastTs).toISOString(),
        medianAmount: medianAmt.toFixed(2),
        latestAmount: latestAmount.toFixed(2),
        currency: g.currency,
        cadence,
        confidence: Math.round(confidence * 100) / 100,
        transactionIds: g.transactionIds,
      });
    }

    candidates.sort((a, b) => {
      if (b.confidence !== a.confidence) return b.confidence - a.confidence;
      return parseFloat(b.latestAmount) - parseFloat(a.latestAmount);
    });

    return { lookbackMonths, candidates: candidates.slice(0, limit) };
  },
};
```

> **Confidence formula note:** The three contributions cap at 0.4 (occurrence count, up to `min_occurrences*1.333`), 0.4 (cadence regularity), and 0.2 (amount stability), summing to a max of 1.0. Irregular cadence yields a 0 regularity contribution, so a candidate with irregular gaps but stable amount still cannot exceed 0.6 — and the 0.45 floor still admits some "stable cost, irregular timing" cases on purpose, per the spec wording.

```typescript
import { z } from "zod";
import { Prisma } from "@prisma/client";
import { prisma } from "../../db.js";
import type { ToolDef } from "./types.js";
import { basicPeriodRange, type BasicPeriod } from "./periods.js";

const Args = z.object({
  period: z.enum(["this_week", "this_month", "last_month"]).describe("Period to summarize cashflow."),
  group_by: z
    .enum(["day", "account"])
    .describe("Optional grouping dimension. Defaults to day.")
    .optional(),
});

type Args = z.infer<typeof Args>;

type Group = {
  key: string;
  label: string;
  incomeTotal: string;
  expenseTotal: string;
  netTotal: string;
  count: number;
};

type Result = {
  period: BasicPeriod;
  groupBy: "day" | "account";
  currency: "CAD";
  incomeTotal: string;
  expenseTotal: string;
  netTotal: string;
  excludedNonCadCount: number;
  groups: Group[];
};

export const getCashflowSummary: ToolDef<Args, Result> = {
  name: "getCashflowSummary",
  description: "Summarize CAD income, expenses, and net cashflow for a period, grouped by day or account.",
  args: Args,
  async run(ctx, args) {
    const groupBy = args.group_by ?? "day";
    const { gte, lt } = await basicPeriodRange(ctx.userId, args.period);

    const [rows, excludedNonCadCount] = await Promise.all([
      prisma.transaction.findMany({
        where: {
          workspaceId: ctx.workspaceId,
          removedAt: null,
          currency: "CAD",
          occurredAt: { gte, lt },
        },
        select: {
          amount: true,
          accountId: true,
          account: { select: { id: true, name: true } },
          occurredAt: true,
        },
      }),
      prisma.transaction.count({
        where: {
          workspaceId: ctx.workspaceId,
          removedAt: null,
          currency: { not: "CAD" },
          occurredAt: { gte, lt },
        },
      }),
    ]);

    type Agg = { key: string; label: string; income: Prisma.Decimal; expense: Prisma.Decimal; count: number };
    const buckets = new Map<string, Agg>();
    let totalIncome = new Prisma.Decimal(0);
    let totalExpense = new Prisma.Decimal(0);

    for (const r of rows) {
      const amt = new Prisma.Decimal(r.amount);
      let key: string;
      let label: string;
      if (groupBy === "day") {
        key = r.occurredAt.toISOString().slice(0, 10);
        label = key;
      } else {
        key = r.accountId;
        label = r.account.name;
      }
      let b = buckets.get(key);
      if (!b) {
        b = { key, label, income: new Prisma.Decimal(0), expense: new Prisma.Decimal(0), count: 0 };
        buckets.set(key, b);
      }
      if (amt.greaterThan(0)) {
        b.expense = b.expense.plus(amt);
        totalExpense = totalExpense.plus(amt);
      } else if (amt.lessThan(0)) {
        const abs = amt.abs();
        b.income = b.income.plus(abs);
        totalIncome = totalIncome.plus(abs);
      }
      b.count += 1;
    }

    const groups: Group[] = [...buckets.values()].map((b) => ({
      key: b.key,
      label: b.label,
      incomeTotal: b.income.toFixed(2),
      expenseTotal: b.expense.toFixed(2),
      netTotal: b.income.minus(b.expense).toFixed(2),
      count: b.count,
    }));

    if (groupBy === "day") {
      groups.sort((a, b) => (a.key < b.key ? -1 : a.key > b.key ? 1 : 0));
    } else {
      groups.sort((a, b) => {
        const aAct = Math.abs(parseFloat(a.incomeTotal)) + Math.abs(parseFloat(a.expenseTotal));
        const bAct = Math.abs(parseFloat(b.incomeTotal)) + Math.abs(parseFloat(b.expenseTotal));
        if (bAct !== aAct) return bAct - aAct;
        return a.label.localeCompare(b.label);
      });
    }

    return {
      period: args.period,
      groupBy,
      currency: "CAD",
      incomeTotal: totalIncome.toFixed(2),
      expenseTotal: totalExpense.toFixed(2),
      netTotal: totalIncome.minus(totalExpense).toFixed(2),
      excludedNonCadCount,
      groups,
    };
  },
};
```
